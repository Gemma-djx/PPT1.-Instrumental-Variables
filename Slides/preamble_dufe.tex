% ---------- Image full-frame command -----------------------
\newcommand{\imageframe}[1]{%
    \begin{frame}[plain]
        \begin{tikzpicture}[remember picture, overlay]
            \node[at=(current page.center)] {%
                \includegraphics[keepaspectratio,
                    width=\paperwidth, height=\paperheight]{#1}
            };
        \end{tikzpicture}
    \end{frame}%
}

% ---------- Transition frame --------------------------------

    \end{frame}%
}
