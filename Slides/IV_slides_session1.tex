\section{Foundations of Instrumental Variables}
%=============================================================

\begin{frame}{Three Conditions for a Valid Instrument}
A valid instrument $Z_i$ must satisfy three conditions:
\begin{enumerate}
  \item \textbf{Relevance (First Stage):} $Z_i$ must be correlated with the
        endogenous treatment $D_i$
        \begin{itemize}
          \item Testable: first-stage F-statistic (rule of thumb: $F>10$)
        \end{itemize}
  \item \textbf{Independence (As-Good-As-Random Assignment):} $Z_i$ must be
        uncorrelated with unobserved confounders $\varepsilon_i$
        \begin{itemize}
          \item Not directly testable; requires economic reasoning and
                robustness checks
        \end{itemize}
  \item \textbf{Exclusion Restriction:} $Z_i$ affects outcome $Y_i$
        \emph{only} through its effect on $D_i$
        \begin{itemize}
          \item No direct effect; testable indirectly via placebo/falsification
        \end{itemize}
\end{enumerate}
\medskip
In Session~2 we add a 4th condition: \textbf{Monotonicity} (for the
LATE interpretation).
\end{frame}

\begin{frame}{Parameters, Estimands, and Estimators}
\vspace{-0.2cm}
Three distinct objects, not always clearly distinguished:
{\small
\begin{itemize}
  \item \bgParam{Parameters} come from economic (or other) models of the world
  \begin{itemize}\itemsep0em
    \item E.g.\ a potential-outcome model relating schooling to earnings
    \item They set the \emph{target} for empirical analyses: what we want to know
  \end{itemize}
  \item \bgEstim{Estimands} are functions of the population data distribution
  \begin{itemize}\itemsep0em
    \item E.g.\ a difference in means or ratio of population regression coefficients
    \item Identification: assumptions that link parameters to estimands
  \end{itemize}
  \item \bgEstor{Estimators} are functions of observed (sample) data
  \begin{itemize}\itemsep0em
    \item E.g.\ a sample ratio of OLS coefficients
    \item Estimators are random variables; statistical inference studies their distribution
  \end{itemize}
\end{itemize}
}
\end{frame}

\begin{frame}{The Endogeneity Problem}
Simple causal model: $Y_i = \violetRed{\beta}\,D_i + \varepsilon_i$, where $D_i$ is the
treatment, $\varepsilon_i$ is the unobserved term.

\medskip
In large samples, OLS converges to:
\[
\pictonBlue{\beta^{OLS}} = \violetRed{\beta} +
  \frac{Cov(\varepsilon_i,D_i)}{Var(D_i)}
\]
\begin{itemize}
  \item OLS identifies $\violetRed{\beta}$ \emph{only if} $Cov(\varepsilon_i,D_i)=0$
  \item \textbf{Selection bias}: units with higher/lower $\varepsilon_i$ select into
        treatment, so $Cov(\varepsilon_i,D_i)\ne0$
\end{itemize}

\smallskip
\textbf{Causal diagram (endogeneity):}
\begin{adjustbox}{width=0.42\textwidth,center}
\begin{tikzpicture}
  \node[state] (D) at (0,0) {$D$};
  \node[state] (Y) at (2,0) {$Y$};
  \node[state,red!70!black] (eps) at (1,1.4) {$\varepsilon$};
  \path[dufenavy,thick] (D) edge node[above,el]{$\violetRed{\beta}$} (Y);
  \path[red!70!black,thick] (eps) edge (Y);
  \path[red!70!black,thick] (eps) edge (D);
\end{tikzpicture}
\end{adjustbox}
\end{frame}

\begin{frame}{The IV Solution}
Suppose $Z_i\in\{0,1\}$ is a binary instrument (e.g.\ lottery assignment).
\begin{itemize}
  \item \textbf{Randomness}: $Cov(Z_i,\varepsilon_i)=0$
        (no direct link to unobservables)
  \item \textbf{Exclusion}: $Z_i$ affects $Y_i$ only through $D_i$
\end{itemize}
\bigskip
Plugging in $\varepsilon_i=Y_i-\violetRed{\beta}D_i$:
\[
Cov(Z_i,Y_i-\violetRed{\beta}D_i)=0
\;\implies\;
\violetRed{\beta} = \frac{Cov(Z_i,Y_i)}{Cov(Z_i,D_i)}
\equiv \pictonBlue{\beta^{IV}},
\]
so long as $Cov(Z_i,D_i)\ne0$ (relevance).\medskip

\textbf{Causal diagram (IV adds $Z$):}
\begin{adjustbox}{width=0.62\textwidth,center}
\begin{tikzpicture}
  \node[state] (D) at (0,0) {$D$};
  \node[state] (Y) at (2,0) {$Y$};
  \node[state,red!70!black] (eps) at (1,1.4) {$\varepsilon$};
  \node[state,dufeteal] (Z) at (-2,0) {$Z$};
  \path[dufenavy,thick] (D) edge node[above,el]{$\violetRed{\beta}$} (Y);
  \path[red!70!black,thick] (eps) edge (Y);
  \path[red!70!black,thick] (eps) edge (D);
  \path[dufeteal,thick] (Z) edge (D);
\end{tikzpicture}
\end{adjustbox}

No arrow from $Z$ to $\varepsilon$ (independence) and no arrow from $Z$
to $Y$ (exclusion).
\end{frame}

\begin{frame}{Reduced Form and First Stage}
A simple decomposition links IV back to familiar OLS regressions:
\[
\pictonBlue{\beta^{IV}}
 = \frac{Cov(Z_i,Y_i)}{Cov(Z_i,D_i)}
 = \frac{Cov(Z_i,Y_i)/Var(Z_i)}{Cov(Z_i,D_i)/Var(Z_i)}
 \equiv \frac{\pictonBlue{\rho}}{\pictonBlue{\pi}}
\]
where $\pictonBlue{\rho}$ and $\pictonBlue{\pi}$ come from two OLS regressions:
\begin{align*}
Y_i &= \kappa + \pictonBlue{\rho}\,Z_i + \nu_i
    \quad \text{(``reduced form'')} \\
D_i &= \mu  + \pictonBlue{\pi}\,Z_i + \eta_i
    \quad \text{(``first stage'')}
\end{align*}
\begin{itemize}
  \item The \textbf{first stage} shows how strongly $Z_i$ moves $D_i$
  \item The \textbf{reduced form} shows the net effect of $Z_i$ on $Y_i$
  \item IV scales the reduced form by the first stage to recover $\violetRed{\beta}$
\end{itemize}
\end{frame}

\begin{frame}{Adding Controls}
If $Z_i$ is as-good-as-random only \emph{conditional} on covariates $W_i$,
add them to both equations:
\begin{align*}
Y_i &= \kappa + \rho Z_i + W_i^\prime\phi  + \nu_i \\
D_i &= \mu  + \pi Z_i  + W_i^\prime\psi + \eta_i
\end{align*}
\begin{itemize}
  \item The Frisch-Waugh-Lovell theorem tells us the effective instrument is
        now $\tilde{Z}_i$: the residuals from regressing $Z_i$ on $W_i$
  \item If $W_i$ is a set of randomization-strata dummies, $\tilde{Z}_i$
        captures within-strata variation in $Z_i$
  \item Key: controls should be chosen on substantive grounds, not to
        maximize the first-stage F-statistic
\end{itemize}
\end{frame}

\begin{frame}{Classic Example: Angrist (1990) Vietnam Draft Lottery}
Angrist (1990, AER) estimated the earnings effect of military service using
Vietnam-era draft-eligibility as an instrument.
\begin{itemize}
  \item $Z_i\in\{0,1\}$: randomly assigned draft eligibility (lottery number)
  \item $D_i\in\{0,1\}$: actual military service
  \item $Y_i$: later-life earnings
\end{itemize}
\medskip
With binary $Z_i$, the IV estimand simplifies to the \textbf{Wald estimand}:
\[
\pictonBlue{\beta^{IV}}
  = \frac{E[Y_i\mid Z_i=1]-E[Y_i\mid Z_i=0]}
         {E[D_i\mid Z_i=1]-E[D_i\mid Z_i=0]}
\]
\begin{itemize}
  \item Numerator: earnings effect of draft eligibility (reduced form)
  \item Denominator: effect of eligibility on actual service probability
        (first stage)
  \item Finding: military service reduces early-career earnings by
        $\approx 15\%$
\end{itemize}
\end{frame}

\section{2SLS and Overidentification}
%=============================================================

\begin{frame}{Multiple Instruments: Overidentification}
When $\dim(Z_i)=L > J=\dim(X_i)$, the system is \emph{overidentified}:
\[
Cov(\tilde{Z}_i,Y_i-X_i^\prime\beta)=0
\;\implies\;
\underbrace{Cov(\tilde{Z}_i,Y_i)}_{L\times 1}
=\underbrace{Cov(\tilde{Z}_i,X_i^\prime)}_{L\times J}
\underbrace{\beta}_{J\times 1}
\]
\begin{itemize}
  \item We have more moment conditions than parameters
  \item Any full-rank $J\times L$ matrix $M$ gives a valid IV estimator:\\
        $\beta^{IV}(M)=\bigl(M\,Cov(\tilde{Z}_i,X_i^\prime)\bigr)^{-1}
        M\,Cov(\tilde{Z}_i,Y_i)$
  \item Choice of $M$ trades off efficiency vs.\ interpretability
\end{itemize}
\end{frame}

\begin{frame}{Two-Stage Least Squares (2SLS)}
2SLS chooses $M^\prime=\pi$ (the matrix of first-stage coefficients):
\begin{itemize}
  \item Effective instrument: $\tilde{Z}_i^*=\tilde{Z}_i^\prime\pi$
        (residualized first-stage fitted values)
  \item Combine instruments according to their predictive power for $X_i$
        $\Rightarrow$ minimizes the 2SLS standard error
\end{itemize}
\medskip
\textbf{Two-stage interpretation:}
\begin{enumerate}
  \item \emph{First stage}: regress $X_i$ on $Z_i$ and $W_i$; save fitted values
        $\hat{X}_i$
  \item \emph{Second stage}: regress $Y_i$ on $\hat{X}_i$ and $W_i$
\end{enumerate}
\begin{alertblock}{Practical warning}
Never run 2SLS by hand---the second-stage standard errors are wrong.
Use \texttt{ivreg2} (Stata) or \texttt{AER::ivreg()} / \texttt{fixest::feols()} (R).
\end{alertblock}
\end{frame}

\begin{frame}{2SLS as a Weighted Average of Just-Identified IVs}
When there is a single treatment ($J=1$), 2SLS has an elegant
weighted-average representation:
\[
\beta^{2SLS} = \sum_\ell \omega_\ell\,\beta_\ell^{IV},
\quad
\omega_\ell = \frac{\pi_\ell^2 Var(\tilde{Z}_{i\ell})}
                   {\sum_{\ell^\prime}\pi_{\ell^\prime}^2 Var(\tilde{Z}_{i\ell^\prime})}
\]
\begin{itemize}
  \item $\beta_\ell^{IV}=\rho_\ell/\pi_\ell$ is the just-identified IV estimate
        using instrument $\ell$ alone
  \item Weights $\omega_\ell$ are proportional to each instrument's contribution
        to first-stage variation
  \item Instruments with stronger first stages receive larger weights
  \item This ``visual IV'' representation (slope of $\rho_\ell$ on $\pi_\ell$)
        is also useful for diagnosis
\end{itemize}
\end{frame}

\begin{frame}{Overidentification Tests}
Under the \emph{constant-effects} model, all just-identified IVs identify the
same $\beta$:
\begin{itemize}
  \item If $\beta_\ell^{IV}$ differ significantly across instruments, something
        is wrong
  \item The Sargan--Hansen statistic tests $H_0: \beta_\ell^{IV}=\beta$ for
        all $\ell$; asymptotically $\chi^2(L-J)$ under the null
\end{itemize}
\medskip
\textbf{Cautions:}
\begin{itemize}
  \item Low power when individual $\hat\beta_\ell^{IV}$ are noisy
  \item Rejection does \emph{not} tell you which instrument is invalid
  \item Under \emph{heterogeneous} effects (Session~2), two valid IVs can
        legitimately have different estimands; the test is not interpretable
  \item Use the test as one input among many, not as a pass/fail criterion
\end{itemize}
\end{frame}

\section{Instrument Strength}
%=============================================================

\begin{frame}{Weak Instruments}
If $\pi$ is small relative to its standard error, the instrument is ``weak'':
\begin{itemize}
  \item Rule of thumb: first-stage $F<10$ signals potential weakness
        (Staiger and Stock, 1997)
  \item \textbf{Bias}: 2SLS is biased toward OLS in small samples;
        the bias is proportional to $1/F$ approximately
  \item \textbf{Size distortion}: standard confidence intervals may
        under-cover in finite samples
\end{itemize}
\medskip
\textbf{Modern guidance (Angrist and Koles\'ar 2022):}
\begin{itemize}
  \item In the just-identified case, 2SLS is \emph{median-unbiased}; the
        SE increase tends to ``cover up'' the bias
  \item Use \texttt{weakivtest} (Stata; Montiel Olea and Pflueger 2015)
        for more nuanced critical values
  \item If $F$ is small: consider Anderson--Rubin or JIVE as robust alternatives
\end{itemize}
\end{frame}

\begin{frame}{Many-IV Bias}
Overidentified 2SLS faces an additional problem when there are many weak instruments:
\begin{itemize}
  \item \textbf{Symptom}: low first-stage F even though SEs are small (more IVs
        $\Rightarrow$ richer first stage $\Rightarrow$ better fit of $D_i$)
  \item \textbf{Mechanism}: overfitting the first stage essentially recreates
        the endogenous variation in $D_i$, biasing 2SLS toward OLS
  \item \textbf{Historical case}: Angrist--Krueger (1991) QoB IV interacted with
        state $\times$ year fixed effects created hundreds of instruments
\end{itemize}
\medskip

\begin{frame}{Practical Guidance on Instrument Strength}
\textbf{Prevention:}
\begin{itemize}
  \item Prefer parsimonious instrument sets; avoid mechanical many-IV settings
  \item If using a Bartik/shift-share IV with many industries, the
        \emph{effective} number of instruments is the number of industries
        (addressed in Session~2)
\end{itemize}
\textbf{Diagnosis:}
\begin{itemize}
  \item Report first-stage F-statistic (or Kleibergen--Paap for clustered SEs)
  \item Use \code{weakivtest} in Stata for heteroskedasticity-robust critical values
  \item See also Lee et al.\ (2020) for $t$-ratio-based inference
\end{itemize}
\textbf{If the instrument is weak:}
\begin{itemize}
  \item Try a more informative functional form for $Z_i$
  \item Collapse to fewer instruments
  \item Use Anderson--Rubin confidence sets (valid regardless of strength)
\end{itemize}
\end{frame}

\section{Classic Applications: Top-5 Journal Papers}
%=============================================================

\begin{frame}{Overview: Two Landmark IV Papers (Post-2010, Top-5 Journals)}
We study two highly cited papers that use \textbf{non-shift-share} IV designs
to establish causal relationships:
\begin{enumerate}
  \item \textbf{Maestas, Mullen, and Strand (2013, \emph{AER})}\\
        ``Does Disability Insurance Receipt Discourage Work?''\\
        {\small IV: randomly assigned Social Security disability examiners.
        Context: US disability insurance applicants.}
  \medskip
  \item \textbf{Aizer and Doyle (2015, \emph{QJE})}\\
        ``Juvenile Incarceration, Human Capital, and Future Crime''\\
        {\small IV: randomly assigned juvenile court judges.
        Context: US juvenile criminal justice system.}
\end{enumerate}
\medskip
Both papers use \textbf{random assignment of decision-makers} as the
identifying instrument---a design that is transparent, credible, and
widely applicable in policy evaluation contexts.
\end{frame}

% ------ Paper 1: Maestas, Mullen & Strand (2013) ------
\begin{frame}{Paper~1: Maestas, Mullen, and Strand (2013, \emph{AER})}
\begin{block}{Citation}
  Maestas, N., Mullen, K.J., and Strand, A.\ (2013).
  ``Does Disability Insurance Receipt Discourage Work? Using Examiner
  Assignment to Estimate Causal Effects of SSDI Receipt.''
  \textit{American Economic Review}, 103(5): 1797--1829.
  {\small($\approx$1{,}300 Google Scholar citations)}
\end{block}

\textbf{Research question:} Does receiving Social Security Disability
Insurance (SSDI) benefits \emph{cause} a reduction in employment?

\medskip
\textbf{Context \& motivation:}
\begin{itemize}
  \item SSDI provides income support to workers with disabling conditions;
        by 2010 over 8.2 million Americans received benefits (\$119 billion)
  \item Key policy concern: does SSDI create a ``poverty trap'' by
        discouraging recipients from returning to work?
  \item OLS is severely biased: healthier rejected applicants would work
        anyway; sicker accepted applicants could not work regardless
        (\emph{reverse causality + selection on unobserved health})
\end{itemize}
\end{frame}

\begin{frame}{MMS (2013): Examiner Assignment IV Strategy}
\textbf{The IV:} Randomly assigned disability examiners differ
systematically in their allowance rates (``leniency''):
\begin{itemize}
  \item SSDI applications are assigned to examiners quasi-randomly
        within local Disability Determination Services (DDS) offices
  \item Examiners vary substantially in their allowance rates even
        after controlling for case characteristics
  \item An examiner's average grant rate (leave-one-out, in other cases)
        is used as the instrument for the applicant's SSDI receipt
\end{itemize}
\medskip
\textbf{IV validity:}
\begin{itemize}
  \item \textbf{Relevance}: examiner leniency strongly predicts own award
        probability (first-stage $F > 100$)
  \item \textbf{Independence}: assignment is quasi-random within offices
        (applicants cannot choose their examiner)
  \item \textbf{Exclusion}: examiner leniency affects employment only through
        SSDI award, not through other channels
  \item \textbf{Monotonicity}: a stricter examiner can only reduce (never
        increase) an applicant's probability of receiving SSDI
\end{itemize}
\end{frame}

\begin{frame}{MMS (2013): Data and Specification}
\textbf{Data:}
\begin{itemize}
  \item Social Security Administration administrative records:
        $\approx$1.5 million SSDI applicants, 1997--2004
  \item \textbf{Treatment}: binary indicator for SSDI award (0/1)
  \item \textbf{Outcomes}: employment indicator, earnings
        (linked to earnings records from IRS W-2 forms)
  \item \textbf{Instrument}: leave-one-out examiner allowance rate
        (average grant rate of the assigned examiner on all other cases)
  \item \textbf{Strata}: DDS office $\times$ quarter fixed effects
        (controls for within-office assignment patterns)
\end{itemize}

\medskip
\textbf{Specification:}
\[
Y_{it} = \alpha + \beta\,\text{SSDI}_{i} + W_i^\prime\gamma + \varepsilon_i
\]
\begin{itemize}
  \item 2SLS using examiner leniency as $Z_i$, conditional on DDS-quarter FEs
  \item Robust to controlling for applicant characteristics (age, diagnosis,
        prior earnings) because assignment is as-good-as-random
\end{itemize}
\end{frame}

\begin{frame}{MMS (2013): Results and Robustness}
\textbf{Key findings:}
\begin{itemize}
  \item SSDI receipt reduces employment by $\approx$\textbf{28 percentage
        points} at ages 45--64 (IV estimate much larger than OLS)
  \item About \textbf{23\%} of marginal SSDI recipients (compliers) could
        work: the ``work-capable'' group whose decision is affected by
        examiner assignment
  \item OLS understates the causal effect because rejected applicants
        who remain employed create a spurious positive ``health effect''
\end{itemize}

\medskip
\textbf{Robustness and sensitivity:}
\begin{itemize}
  \item \emph{Placebo test}: examiner leniency does not predict
        pre-application employment trends (balance on pre-period outcomes)
  \item \emph{Alternative examiner measures}: using quintiles of leniency
        instead of the continuous measure gives nearly identical results
  \item \emph{Sub-group consistency}: effects are similar across age groups,
        diagnosis categories, and states
  \item \emph{Complier analysis}: marginal recipients are younger, healthier,
        and have higher prior earnings than average recipients, explaining
        why SSDI receipt suppresses their employment substantially
\end{itemize}
\end{frame}

\begin{frame}{MMS (2013): Contribution and IV Lessons}
\textbf{Academic contributions:}
\begin{itemize}
  \item First credible causal estimate of SSDI's effect on employment;
        resolves a long-standing identification debate
  \item Shows SSDI creates work disincentives for a sizeable share of
        recipients---important for disability insurance design
  \item Introduced the ``examiner leniency'' IV into economics; spawned
        a large literature on disability insurance, screening, and
        physician/administrative judgment
\end{itemize}

\medskip
\textbf{IV methodology lessons:}
\begin{itemize}
  \item \textbf{Random assignment of decision-makers} is a powerful source
        of exogenous variation when direct randomization is infeasible
  \item The \textbf{leave-one-out leniency measure} avoids mechanical
        correlation between one's own case and the instrument
  \item \textbf{LATE interpretation} is transparent: IV estimates the effect
        of SSDI on ``compliers''---those whose award decision is influenced
        by examiner assignment
  \item Strata fixed effects (DDS-quarter) are essential to isolate
        within-office random variation
\end{itemize}
\end{frame}

% ------ Paper 2: Aizer & Doyle (2015) ------
\begin{frame}{Paper~2: Aizer and Doyle (2015, \emph{QJE})}
\begin{block}{Citation}
  Aizer, A.\ and Doyle, J.J., Jr.\ (2015).
  ``Juvenile Incarceration, Human Capital, and Future Crime: Evidence
  from Randomly Assigned Judges.''
  \textit{Quarterly Journal of Economics}, 130(2): 759--803.
  {\small($\approx$900 Google Scholar citations)}
\end{block}

\textbf{Research question:} Does incarcerating juveniles reduce or increase
their subsequent criminal activity and human capital?

\medskip
\textbf{Context \& motivation:}
\begin{itemize}
  \item The US incarcerates approximately 70{,}000 juveniles on any given day;
        recidivism rates are high ($\approx$70\% re-arrested within 3 years)
  \item Policy debate: does incarceration deter future crime (deterrence),
        or does it increase it via criminal networks and disrupted schooling?
  \item OLS is biased: more serious offenders are both more likely to be
        incarcerated \emph{and} more likely to re-offend regardless
\end{itemize}
\end{frame}

\begin{frame}{Aizer \& Doyle (2015): Judge IV Strategy}
\textbf{The IV:} Randomly assigned juvenile court judges vary in their
propensity to incarcerate:
\begin{itemize}
  \item Juvenile cases are assigned to judges \emph{randomly} within
        courthouses in Chicago, Illinois
  \item Judges differ substantially in their incarceration rates even
        for similar offense types and defendant characteristics
  \item A judge's leave-one-out incarceration rate on other cases is the
        instrument for whether defendant $i$ is incarcerated
\end{itemize}
\medskip
\textbf{IV validity:}
\begin{itemize}
  \item \textbf{Relevance}: judge leniency strongly predicts incarceration
        (first-stage $F \approx 35$--50)
  \item \textbf{Independence}: assignment is explicitly random within court
        (confirmed by balance tests on pre-determined characteristics)
  \item \textbf{Exclusion}: judge assignment affects outcomes only through
        the incarceration decision, not through other channels
  \item \textbf{Monotonicity}: a stricter judge cannot reduce a defendant's
        incarceration probability (one-directional instrument)
\end{itemize}
\end{frame}

\begin{frame}{Aizer \& Doyle (2015): Data and Specification}
\textbf{Data:}
\begin{itemize}
  \item Illinois Juvenile Court data linked to adult criminal records,
        high school graduation records, and mortality data; 1990--2006
  \item $\approx$35{,}000 juvenile cases in Chicago (ages 10--17)
  \item \textbf{Treatment}: binary incarceration indicator (0/1)
  \item \textbf{Outcomes}: adult incarceration, high school graduation,
        employment, and earnings in adulthood (ages 25--35)
  \item \textbf{Instrument}: leave-one-out judge incarceration rate
        (within courthouse, controlling for offense type and year)
\end{itemize}

\medskip
\textbf{Specification:}
\[
Y_i = \alpha + \beta\,\text{Incarcerated}_i + W_i^\prime\gamma + \varepsilon_i
\]
\begin{itemize}
  \item 2SLS using judge leniency as $Z_i$, with courthouse-year-offense
        category fixed effects as strata
  \item Outcomes measured 5--15 years after the juvenile offense
\end{itemize}
\end{frame}

\begin{frame}{Aizer \& Doyle (2015): Results and Robustness}
\textbf{Key findings:}
\begin{itemize}
  \item Juvenile incarceration \emph{reduces} high school graduation
        by $\approx$\textbf{13 percentage points}
  \item Juvenile incarceration \emph{increases} adult incarceration
        probability by $\approx$\textbf{22--26 percentage points}
  \item Earnings are reduced significantly; effects persist for decades
  \item Net cost-benefit analysis: incarceration costs society roughly
        \$1.1--1.4 million per juvenile (present value), far exceeding
        any deterrence benefit
\end{itemize}

\medskip
\textbf{Robustness and sensitivity:}
\begin{itemize}
  \item \emph{Balance test}: judge leniency is uncorrelated with defendant
        race, prior offenses, and offense severity (random assignment holds)
  \item \emph{Placebo}: judge leniency does not predict pre-offense
        outcomes (no pre-existing differences across complier groups)
  \item \emph{Different charge types}: results hold across assault, drug,
        property, and public order offenses
  \item \emph{Bandwidth variation}: narrowing the leave-one-out window
        (same offense type only) gives similar results
\end{itemize}
\end{frame}

\begin{frame}{Aizer \& Doyle (2015): Contribution and IV Lessons}
\textbf{Academic contributions:}
\begin{itemize}
  \item First causal evidence that juvenile incarceration \emph{increases}
        adult criminality and reduces human capital---a critical finding for
        criminal justice reform debates
  \item Demonstrates that the criminal justice system can perpetuate rather
        than reduce criminal behavior through incapacitation effects
  \item Extended the judge-leniency IV approach (Kling 2006) to juvenile
        settings and linked to long-run human capital outcomes
\end{itemize}

\medskip
\textbf{IV methodology lessons:}
\begin{itemize}
  \item \textbf{Judge/examiner leniency IV} is now a standard tool whenever
        administrative decisions are quasi-randomly assigned
  \item Key to validity: \emph{random} (not quasi-random) assignment within
        strata is what makes the exclusion credible
  \item \textbf{LATE}: IV identifies the effect among ``compliers''---juveniles
        whose incarceration decision is swayed by judge assignment, typically
        those near the incarceration margin
  \item Long-run outcomes require linking administrative datasets across
        agencies (courts, schools, IRS, corrections)
\end{itemize}
\end{frame}

\begin{frame}{Session 1 Summary}
\textbf{IV Foundations:}
\begin{itemize}
  \item IV validity requires relevance, independence, and exclusion
  \item $\beta^{IV}=\rho/\pi$ (reduced form $\div$ first stage)
  \item With controls: effective instrument $\tilde Z_i$ = residual of
        $Z_i$ on $W_i$ (Frisch--Waugh)
\end{itemize}

\medskip
\textbf{2SLS and Overidentification:}
\begin{itemize}
  \item 2SLS = efficiency-optimal combination of multiple instruments
  \item 2SLS = weighted average of just-identified IVs
  \item Sargan--Hansen test: diagnostic, not a ``certification of validity''
\end{itemize}

\medskip
\textbf{Instrument Strength:}
\begin{itemize}
  \item $F>10$ rule of thumb; \texttt{weakivtest} for robust critical values
  \item Many-IV bias: overfitting first stage recreates endogeneity
\end{itemize}

\medskip
\textbf{Applications (Top-5, Post-2010):}
\begin{itemize}
  \item Maestas, Mullen \& Strand (2013, AER): examiner leniency IV
        $\Rightarrow$ SSDI receipt reduces employment by 28pp
  \item Aizer \& Doyle (2015, QJE): judge leniency IV $\Rightarrow$
        juvenile incarceration reduces graduation and increases re-offending
\end{itemize}
\end{frame}
