\section{Foundations of Instrumental Variables}
%=============================================================

\begin{frame}{Three Conditions for a Valid Instrument}
A valid instrument $Z_i$ must satisfy three conditions:
\begin{enumerate}
  \item \textbf{Relevance (First Stage):} $Z_i$ must be correlated with the
        endogenous treatment $D_i$
        \begin{itemize}
          \item Testable: first-stage F-statistic (rule of thumb: $F>10$)
        \end{itemize}
  \item \textbf{Independence (As-Good-As-Random Assignment):} $Z_i$ must be
        uncorrelated with unobserved confounders $\varepsilon_i$
        \begin{itemize}
          \item Not directly testable; requires economic reasoning and
                robustness checks
        \end{itemize}
  \item \textbf{Exclusion Restriction:} $Z_i$ affects outcome $Y_i$
        \emph{only} through its effect on $D_i$
        \begin{itemize}
          \item No direct effect; testable indirectly via placebo/falsification
        \end{itemize}
\end{enumerate}
\medskip
In Session~2 we add a 4th condition: \textbf{Monotonicity} (for the
LATE interpretation).
\end{frame}

\begin{frame}{Parameters, Estimands, and Estimators}
\vspace{-0.2cm}
Three distinct objects, not always clearly distinguished:
{\small
\begin{itemize}
  \item \bgParam{Parameters} come from economic (or other) models of the world
  \begin{itemize}\itemsep0em
    \item E.g.\ a potential-outcome model relating schooling to earnings
    \item They set the \emph{target} for empirical analyses: what we want to know
  \end{itemize}
  \item \bgEstim{Estimands} are functions of the population data distribution
  \begin{itemize}\itemsep0em
    \item E.g.\ a difference in means or ratio of population regression coefficients
    \item Identification: assumptions that link parameters to estimands
  \end{itemize}
  \item \bgEstor{Estimators} are functions of observed (sample) data
  \begin{itemize}\itemsep0em
    \item E.g.\ a sample ratio of OLS coefficients
    \item Estimators are random variables; statistical inference studies their distribution
  \end{itemize}
\end{itemize}
}
\end{frame}

\begin{frame}{The Endogeneity Problem}
Simple causal model: $Y_i = \violetRed{\beta}\,D_i + \varepsilon_i$, where $D_i$ is the
treatment, $\varepsilon_i$ is the unobserved term.

\bigskip
In large samples, OLS converges to:
\[
\pictonBlue{\beta^{OLS}} = \violetRed{\beta} +
  \frac{Cov(\varepsilon_i,D_i)}{Var(D_i)}
\]
\begin{itemize}
  \item OLS identifies $\violetRed{\beta}$ \emph{only if} $Cov(\varepsilon_i,D_i)=0$
  \item \textbf{Selection bias}: units with higher/lower $\varepsilon_i$ select into
        treatment, so $Cov(\varepsilon_i,D_i)\ne0$
\end{itemize}

\bigskip
\textbf{Causal diagram:}
\begin{adjustbox}{width=0.55\textwidth,center}
\begin{tikzpicture}
  \node[state] (D) at (0,0) {$D$};
  \node[state] (Y) at (2,0) {$Y$};
  \node[state,red!70!black] (eps) at (1,1.4) {$\varepsilon$};
  \path[dufenavy,thick] (D) edge node[above,el]{$\violetRed{\beta}$} (Y);
  \path[red!70!black,thick] (eps) edge (Y);
  \path[red!70!black,thick] (eps) edge (D);
\end{tikzpicture}
\end{adjustbox}
\end{frame}

\begin{frame}{The IV Solution}
Suppose $Z_i\in\{0,1\}$ is a binary instrument (e.g.\ lottery assignment).
\begin{itemize}
  \item \textbf{Randomness}: $Cov(Z_i,\varepsilon_i)=0$
        (no direct link to unobservables)
  \item \textbf{Exclusion}: $Z_i$ affects $Y_i$ only through $D_i$
\end{itemize}
\bigskip
Plugging in $\varepsilon_i=Y_i-\violetRed{\beta}D_i$:
\[
Cov(Z_i,Y_i-\violetRed{\beta}D_i)=0
\;\implies\;
\violetRed{\beta} = \frac{Cov(Z_i,Y_i)}{Cov(Z_i,D_i)}
\equiv \pictonBlue{\beta^{IV}},
\]
so long as $Cov(Z_i,D_i)\ne0$ (relevance).\medskip

\textbf{Causal diagram (IV adds $Z$):}
\begin{adjustbox}{width=0.62\textwidth,center}
\begin{tikzpicture}
  \node[state] (D) at (0,0) {$D$};
  \node[state] (Y) at (2,0) {$Y$};
  \node[state,red!70!black] (eps) at (1,1.4) {$\varepsilon$};
  \node[state,dufeteal] (Z) at (-2,0) {$Z$};
  \path[dufenavy,thick] (D) edge node[above,el]{$\violetRed{\beta}$} (Y);
  \path[red!70!black,thick] (eps) edge (Y);
  \path[red!70!black,thick] (eps) edge (D);
  \path[dufeteal,thick] (Z) edge (D);
\end{tikzpicture}
\end{adjustbox}

No arrow from $Z$ to $\varepsilon$ (independence) and no arrow from $Z$
to $Y$ (exclusion).
\end{frame}

\begin{frame}{Reduced Form and First Stage}
A simple decomposition links IV back to familiar OLS regressions:
\[
\pictonBlue{\beta^{IV}}
 = \frac{Cov(Z_i,Y_i)}{Cov(Z_i,D_i)}
 = \frac{Cov(Z_i,Y_i)/Var(Z_i)}{Cov(Z_i,D_i)/Var(Z_i)}
 \equiv \frac{\pictonBlue{\rho}}{\pictonBlue{\pi}}
\]
where $\pictonBlue{\rho}$ and $\pictonBlue{\pi}$ come from two OLS regressions:
\begin{align*}
Y_i &= \kappa + \pictonBlue{\rho}\,Z_i + \nu_i
    \quad \text{(``reduced form'')} \\
D_i &= \mu  + \pictonBlue{\pi}\,Z_i + \eta_i
    \quad \text{(``first stage'')}
\end{align*}
\begin{itemize}
  \item The \textbf{first stage} shows how strongly $Z_i$ moves $D_i$
  \item The \textbf{reduced form} shows the net effect of $Z_i$ on $Y_i$
  \item IV scales the reduced form by the first stage to recover $\violetRed{\beta}$
\end{itemize}
\end{frame}

\begin{frame}{Adding Controls}
If $Z_i$ is as-good-as-random only \emph{conditional} on covariates $W_i$,
add them to both equations:
\begin{align*}
Y_i &= \kappa + \rho Z_i + W_i^\prime\phi  + \nu_i \\
D_i &= \mu  + \pi Z_i  + W_i^\prime\psi + \eta_i
\end{align*}
\begin{itemize}
  \item The Frisch-Waugh-Lovell theorem tells us the effective instrument is
        now $\tilde{Z}_i$: the residuals from regressing $Z_i$ on $W_i$
  \item If $W_i$ is a set of randomization-strata dummies, $\tilde{Z}_i$
        captures within-strata variation in $Z_i$
  \item Key: controls should be chosen on substantive grounds, not to
        maximize the first-stage F-statistic
\end{itemize}
\end{frame}

\begin{frame}{Classic Example: Angrist (1990) Vietnam Draft Lottery}
Angrist (1990, AER) estimated the earnings effect of military service using
Vietnam-era draft-eligibility as an instrument.
\begin{itemize}
  \item $Z_i\in\{0,1\}$: randomly assigned draft eligibility (lottery number)
  \item $D_i\in\{0,1\}$: actual military service
  \item $Y_i$: later-life earnings
\end{itemize}
\medskip
With binary $Z_i$, the IV estimand simplifies to the \textbf{Wald estimand}:
\[
\pictonBlue{\beta^{IV}}
  = \frac{E[Y_i\mid Z_i=1]-E[Y_i\mid Z_i=0]}
         {E[D_i\mid Z_i=1]-E[D_i\mid Z_i=0]}
\]
\begin{itemize}
  \item Numerator: earnings effect of draft eligibility (reduced form)
  \item Denominator: effect of eligibility on actual service probability
        (first stage)
  \item Finding: military service reduces early-career earnings by
        $\approx 15\%$
\end{itemize}
\end{frame}

\section{2SLS and Overidentification}
%=============================================================

\begin{frame}{Multiple Instruments: Overidentification}
When $\dim(Z_i)=L > J=\dim(X_i)$, the system is \emph{overidentified}:
\[
Cov(\tilde{Z}_i,Y_i-X_i^\prime\beta)=0
\;\implies\;
\underbrace{Cov(\tilde{Z}_i,Y_i)}_{L\times 1}
=\underbrace{Cov(\tilde{Z}_i,X_i^\prime)}_{L\times J}
\underbrace{\beta}_{J\times 1}
\]
\begin{itemize}
  \item We have more moment conditions than parameters
  \item Any full-rank $J\times L$ matrix $M$ gives a valid IV estimator:\\
        $\beta^{IV}(M)=\bigl(M\,Cov(\tilde{Z}_i,X_i^\prime)\bigr)^{-1}
        M\,Cov(\tilde{Z}_i,Y_i)$
  \item Choice of $M$ trades off efficiency vs.\ interpretability
\end{itemize}
\end{frame}

\begin{frame}{Two-Stage Least Squares (2SLS)}
2SLS chooses $M^\prime=\pi$ (the matrix of first-stage coefficients):
\begin{itemize}
  \item Effective instrument: $\tilde{Z}_i^*=\tilde{Z}_i^\prime\pi$
        (residualized first-stage fitted values)
  \item Combine instruments according to their predictive power for $X_i$
        $\Rightarrow$ minimizes the 2SLS standard error
\end{itemize}
\medskip
\textbf{Two-stage interpretation:}
\begin{enumerate}
  \item \emph{First stage}: regress $X_i$ on $Z_i$ and $W_i$; save fitted values
        $\hat{X}_i$
  \item \emph{Second stage}: regress $Y_i$ on $\hat{X}_i$ and $W_i$
\end{enumerate}
\begin{alertblock}{Practical warning}
Never run 2SLS by hand---the second-stage standard errors are wrong.
Use \texttt{ivreg2} (Stata) or \texttt{AER::ivreg()} / \texttt{fixest::feols()} (R).
\end{alertblock}
\end{frame}

\begin{frame}{2SLS as a Weighted Average of Just-Identified IVs}
When there is a single treatment ($J=1$), 2SLS has an elegant
weighted-average representation:
\[
\beta^{2SLS} = \sum_\ell \omega_\ell\,\beta_\ell^{IV},
\quad
\omega_\ell = \frac{\pi_\ell^2 Var(\tilde{Z}_{i\ell})}
                   {\sum_{\ell^\prime}\pi_{\ell^\prime}^2 Var(\tilde{Z}_{i\ell^\prime})}
\]
\begin{itemize}
  \item $\beta_\ell^{IV}=\rho_\ell/\pi_\ell$ is the just-identified IV estimate
        using instrument $\ell$ alone
  \item Weights $\omega_\ell$ are proportional to each instrument's contribution
        to first-stage variation
  \item Instruments with stronger first stages receive larger weights
  \item This ``visual IV'' representation (slope of $\rho_\ell$ on $\pi_\ell$)
        is also useful for diagnosis
\end{itemize}
\end{frame}

\begin{frame}{Overidentification Tests}
Under the \emph{constant-effects} model, all just-identified IVs identify the
same $\beta$:
\begin{itemize}
  \item If $\beta_\ell^{IV}$ differ significantly across instruments, something
        is wrong
  \item The Sargan--Hansen statistic tests $H_0: \beta_\ell^{IV}=\beta$ for
        all $\ell$; asymptotically $\chi^2(L-J)$ under the null
\end{itemize}
\medskip
\textbf{Cautions:}
\begin{itemize}
  \item Low power when individual $\hat\beta_\ell^{IV}$ are noisy
  \item Rejection does \emph{not} tell you which instrument is invalid
  \item Under \emph{heterogeneous} effects (Session~2), two valid IVs can
        legitimately have different estimands; the test is not interpretable
  \item Use the test as one input among many, not as a pass/fail criterion
\end{itemize}
\end{frame}

\section{Instrument Strength}
%=============================================================

\begin{frame}{Weak Instruments}
If $\pi$ is small relative to its standard error, the instrument is ``weak'':
\begin{itemize}
  \item Rule of thumb: first-stage $F<10$ signals potential weakness
        (Staiger and Stock, 1997)
  \item \textbf{Bias}: 2SLS is biased toward OLS in small samples;
        the bias is proportional to $1/F$ approximately
  \item \textbf{Size distortion}: standard confidence intervals may
        under-cover in finite samples
\end{itemize}
\medskip
\textbf{Modern guidance (Angrist and Koles\'ar 2022):}
\begin{itemize}
  \item In the just-identified case, 2SLS is \emph{median-unbiased}; the
        SE increase tends to ``cover up'' the bias
  \item Use \texttt{weakivtest} (Stata; Montiel Olea and Pflueger 2015)
        for more nuanced critical values
  \item If $F$ is small: consider Anderson--Rubin or JIVE as robust alternatives
\end{itemize}
\end{frame}

\begin{frame}{Many-IV Bias}
Overidentified 2SLS faces an additional problem when there are many weak instruments:
\begin{itemize}
  \item \textbf{Symptom}: low first-stage F even though SEs are small (more IVs
        $\Rightarrow$ richer first stage $\Rightarrow$ better fit of $D_i$)
  \item \textbf{Mechanism}: overfitting the first stage essentially recreates
        the endogenous variation in $D_i$, biasing 2SLS toward OLS
  \item \textbf{Historical case}: Angrist--Krueger (1991) QoB IV interacted with
        state $\times$ year fixed effects created hundreds of instruments
\end{itemize}
\medskip

\begin{frame}{Practical Guidance on Instrument Strength}
\textbf{Prevention:}
\begin{itemize}
  \item Prefer parsimonious instrument sets; avoid mechanical many-IV settings
  \item If using a Bartik/shift-share IV with many industries, the
        \emph{effective} number of instruments is the number of industries
        (addressed in Session~2)
\end{itemize}
\textbf{Diagnosis:}
\begin{itemize}
  \item Report first-stage F-statistic (or Kleibergen--Paap for clustered SEs)
  \item Use \code{weakivtest} in Stata for heteroskedasticity-robust critical values
  \item See also Lee et al.\ (2020) for $t$-ratio-based inference
\end{itemize}
\textbf{If the instrument is weak:}
\begin{itemize}
  \item Try a more informative functional form for $Z_i$
  \item Collapse to fewer instruments
  \item Use Anderson--Rubin confidence sets (valid regardless of strength)
\end{itemize}
\end{frame}

\section{Classic Applications: Top-5 Journal Papers}
%=============================================================

\begin{frame}{Overview: Two Landmark IV Papers (Post-2010, Top-5 Journals)}
We study two highly cited papers that use creative IV designs to establish causal relationships:
\begin{enumerate}
  \item \textbf{Nunn and Qian (2014, \emph{AER})}\\
        ``US Food Aid and Civil Conflict''\\
        {\small Uses US wheat production $\times$ historical aid patterns as IV.
        Context: Africa and developing world.}
  \medskip
  \item \textbf{Acemoglu and Restrepo (2020, \emph{JPE})}\\
        ``Robots and Employment: Evidence from US Commuting Zones''\\
        {\small Uses Bartik shift-share IV with robot adoption in other countries.
        Context: US labor markets.}
\end{enumerate}
\medskip
Both papers were cited 1{,}000+ times within a decade of publication
and have had substantial impact on empirical methodology.
\end{frame}

% ------ Paper 1: Nunn & Qian (2014) ------
\begin{frame}{Paper~1: Nunn and Qian (2014, \emph{AER})}
\begin{block}{Citation}
  Nunn, N.\ and Qian, N.\ (2014).
  ``US Food Aid and Civil Conflict.''
  \textit{American Economic Review}, 104(6): 1630--1666.
  {\small($\approx$1{,}700 Google Scholar citations)}
\end{block}

\textbf{Research question:} Does US food aid \emph{cause} civil conflict in
recipient countries?

\medskip
\textbf{Context \& motivation:}
\begin{itemize}
  \item The US is the world's largest food-aid donor; annual shipments
        exceed \$1 billion
  \item Theoretical ambiguity: aid may reduce conflict by alleviating
        poverty, or \emph{increase} conflict by providing resources to loot
        or by strengthening rebel groups
  \item OLS correlation is unreliable: countries in conflict receive more
        aid (\emph{reverse causality}); unobserved factors (instability,
        colonial history) drive both variables (\emph{omitted variable bias})
\end{itemize}
\end{frame}

\begin{frame}{Nunn \& Qian (2014): Identification Strategy}
\textbf{The IV:} Interaction of US wheat production with a country's
historical propensity to receive US food aid:
\[
Z_{it} = \underbrace{\text{USWheat}_{t}}_{\text{supply shock}}
         \times
         \underbrace{\bar{P}_i}_{\text{historical aid share}}
\]
\begin{itemize}
  \item $\text{USWheat}_t$: deviations in US wheat production (driven by
        domestic weather and farm policy, \emph{exogenous} to recipient
        conditions)
  \item $\bar{P}_i$: fraction of years country $i$ received US food aid
        in 1971--1974 (pre-determined baseline capturing structural
        factors: colonial ties, Cold War alignment, proximity)
\end{itemize}
\medskip
\textbf{Identifying variation:}
\begin{itemize}
  \item In bad US harvest years, food-aid flows fall globally
  \item Countries with higher historical aid propensity experience larger
        fluctuations---variation that is uncorrelated with their own
        domestic conditions
\end{itemize}
\end{frame}

\begin{frame}{Nunn \& Qian (2014): Data and Specification}
\textbf{Data:}
\begin{itemize}
  \item 125 food-aid recipient countries, annual panel 1971--2006
  \item \textbf{Outcome}: intra-state conflict onset \& incidence
        (from PRIO Armed Conflict Dataset)
  \item \textbf{Treatment}: US food aid shipments (USAID Greenbook)
  \item \textbf{Instrument}: USDA wheat production $\times$ historical
        aid share
  \item \textbf{Controls}: country and year fixed effects; oil prices;
        US GDP growth; country-specific trends
\end{itemize}

\medskip
\textbf{Specification:}
\[
\text{Conflict}_{it} = \alpha_i + \mu_t + \beta\,\text{FoodAid}_{it}
  + \mathbf{W}_{it}^\prime\gamma + \varepsilon_{it}
\]
Instrumenting $\text{FoodAid}_{it}$ with $Z_{it}$ via 2SLS.

\medskip
\textbf{Level of identifying variation:} US wheat production varies
year-by-year; the interaction creates country-year level variation.
\end{frame}

\begin{frame}{Nunn \& Qian (2014): Results and Robustness}
\textbf{Key findings:}
\begin{itemize}
  \item A 10\% increase in food-aid deliveries raises the probability of
        civil conflict onset by $\approx 1.14$ percentage points (large
        relative to an $\approx 5\%$ base rate)
  \item Effects are concentrated in countries with weak initial institutions
        and high ethnic fractionalization
  \item Stronger effects for food aid that arrives \emph{during} conflict
        (suggests rebel looting mechanism)
\end{itemize}

\medskip
\textbf{Robustness and sensitivity:}
\begin{itemize}
  \item \emph{Placebo crops}: using soybean or corn production instead of
        wheat yields near-zero first stages and insignificant reduced
        forms---these crops are not price-supported and less likely to
        drive food-aid volumes
  \item \emph{Alternative instruments}: using only lagged US wheat
        production (dropping $\bar P_i$ variation) gives similar results
  \item \emph{Excluding major recipients}: dropping top food-aid
        recipients (Egypt, Ethiopia) does not change results
  \item \emph{Heterogeneous trends}: allowing country-specific linear
        trends leaves findings robust
\end{itemize}
\end{frame}

\begin{frame}{Nunn \& Qian (2014): Contribution and IV Lessons}
\textbf{Academic contributions:}
\begin{itemize}
  \item First large-$N$ causal evidence that food aid can increase, not
        decrease, conflict---a ``boomerang effect''
  \item Shapes the debate on aid effectiveness (Easterly vs.\ Sachs)
        and humanitarian intervention design
  \item Demonstrates that agricultural commodity shocks in donor
        countries can serve as valid instruments for aid flows
\end{itemize}

\medskip
\textbf{IV methodology lessons:}
\begin{itemize}
  \item The \textbf{interaction instrument} $Z_{it}=\text{Shock}_t\times
        \text{Exposure}_i$ is a workhorse design in development economics
  \item Validity rests on the \emph{supply} side (US weather/farm policy
        unrelated to conflict in recipient countries)
  \item The inclusion of country FE absorbs time-invariant confounders
        correlated with $\bar P_i$; year FE absorbs global shocks
  \item Always test first stage separately by sub-groups and with
        alternative instrument definitions
\end{itemize}
\end{frame}

% ------ Paper 2: Acemoglu & Restrepo (2020) ------
\begin{frame}{Paper~2: Acemoglu and Restrepo (2020, \emph{JPE})}
\begin{block}{Citation}
  Acemoglu, D.\ and Restrepo, P.\ (2020).
  ``Robots and Employment: Evidence from US Commuting Zones, 1990--2007.''
  \textit{Journal of Political Economy}, 128(6): 2188--2244.
  {\small($\approx$1{,}800 Google Scholar citations)}
\end{block}

\textbf{Research question:} What is the causal effect of industrial robot
adoption on local labor markets?

\medskip
\textbf{Context \& motivation:}
\begin{itemize}
  \item Industrial robot density rose sharply in Europe and the US from
        the 1990s---particularly in automotive, electronics, and metals
  \item Economic theory predicts robots displace routine tasks but may
        create complementary jobs elsewhere
  \item OLS is biased: industries undergoing productivity shocks may
        adopt robots \emph{and} have other trends in wages/employment
\end{itemize}
\end{frame}

\begin{frame}{Acemoglu \& Restrepo (2020): Shift-Share IV Design}
\textbf{The instrument:} Robot adoption in other high-income countries
``translated'' to US commuting zones via pre-period industry shares:
\[
Z_i = \sum_k \underbrace{l_{ik,1990}}_{\text{share}}
             \times
             \underbrace{\Delta\text{Robots}_{k,\text{OtherCountries}}}_{\text{shock}}
\]
\begin{itemize}
  \item \textbf{Shocks}: growth in robots per worker in industry $k$
        across Europe (France, Germany, Italy, Finland, Sweden, etc.)
  \item \textbf{Shares}: CZ $i$'s employment share in industry $k$ as
        of 1990 (pre-determined)
\end{itemize}
\medskip
\textbf{Validity argument (parallel to ADH 2013):}
\begin{itemize}
  \item Robot adoption in European industries reflects global technology
        improvements, not US local demand conditions
  \item Pre-period shares capture industrial composition that predates
        the robot wave---plausibly uncorrelated with contemporaneous
        US labor demand shocks
\end{itemize}
\end{frame}

\begin{frame}{Acemoglu \& Restrepo (2020): Data and Specification}
\textbf{Data:}
\begin{itemize}
  \item 722 US commuting zones (CZs), two periods: 1990--2007
  \item \textbf{Treatment}: robots per 1{,}000 workers in each CZ
        (constructed from IFR data mapped to CZ industries)
  \item \textbf{Outcomes}: employment-to-population ratio; log wages;
        share in manufacturing; hours worked
  \item \textbf{Instrument}: exposure to foreign robot adoption via Bartik construction above
  \item \textbf{Controls}: demographics, routine task intensity, CZ
        fixed effects, census division fixed effects
\end{itemize}

\medskip
\textbf{Specification:}
\[
\Delta Y_i = \alpha + \beta\,\Delta\text{Robots}_i
           + X_i^\prime\gamma + \varepsilon_i
\]
Instrumenting $\Delta\text{Robots}_i$ with $Z_i$ via 2SLS across two
long differences (1990--2000, 2000--2007).
\end{frame}

\begin{frame}{Acemoglu \& Restrepo (2020): Results and Robustness}
\textbf{Key findings:}
\begin{itemize}
  \item One additional robot per 1{,}000 workers $\Rightarrow$ employment
        ratio falls by 0.18--0.34 pp and wages by 0.25--0.5\%
  \item Roughly 400{,}000--670{,}000 jobs displaced by robots over
        1990--2007
  \item Effects concentrated in manufacturing; no significant offsetting
        job creation in other sectors
  \item Larger effects for lower-skill, routine-task workers
\end{itemize}

\medskip
\textbf{Robustness and sensitivity:}
\begin{itemize}
  \item \emph{Pre-trend test}: robot exposure does not predict 1970--1990
        employment trends (no pre-existing differences)
  \item \emph{Alternative shares}: using 1970 industry shares (farther
        from the robot wave) gives similar results
  \item \emph{Excluding auto-heavy CZs}: dropping the Midwest
        automotive belt does not materially change estimates
  \item \emph{China competition control}: conditioning on ADH-style
        Chinese import competition exposure leaves robot effects intact
  \item \emph{Clustering}: results robust to CZ, state, and
        industry clustering
\end{itemize}
\end{frame}

\begin{frame}{Acemoglu \& Restrepo (2020): Contribution and IV Lessons}
\textbf{Academic contributions:}
\begin{itemize}
  \item Provides first rigorous causal evidence that robots reduce employment
        and wages---fueling the automation-inequality debate
  \item Methodologically pairs theory (task-based model) with credible
        causal identification
  \item Spawned a large literature on technology, automation, and
        labor market polarization
\end{itemize}

\medskip
\textbf{IV methodology lessons:}
\begin{itemize}
  \item The \textbf{Bartik shift-share} design (industry shares $\times$
        global shocks) is now a standard tool in regional economics---see
        also Borusyak, Hull, Jaravel (2022) for formal conditions
  \item Foreign-country shocks are a natural exogeneity source when
        domestic conditions are confounded
  \item Pre-trend and placebo tests are essential for validating
        the parallel-trends implicit in share-based instruments
  \item SSIV formal theory and China HSR illustration are covered in
        Session~2
\end{itemize}
\end{frame}

\begin{frame}{Session 1 Summary}
\textbf{IV Foundations:}
\begin{itemize}
  \item IV validity requires relevance, independence, and exclusion
  \item $\beta^{IV}=\rho/\pi$ (reduced form $\div$ first stage)
  \item With controls: effective instrument $\tilde Z_i$ = residual of
        $Z_i$ on $W_i$ (Frisch--Waugh)
\end{itemize}

\medskip
\textbf{2SLS and Overidentification:}
\begin{itemize}
  \item 2SLS = efficiency-optimal combination of multiple instruments
  \item 2SLS = weighted average of just-identified IVs
  \item Sargan--Hansen test: diagnostic, not a ``certification of validity''
\end{itemize}

\medskip
\textbf{Instrument Strength:}
\begin{itemize}
  \item $F>10$ rule of thumb; \texttt{weakivtest} for robust critical values
  \item Many-IV bias: overfitting first stage recreates endogeneity
\end{itemize}

\medskip
\textbf{Applications (Top-5, Post-2010):}
\begin{itemize}
  \item Nunn \& Qian (2014, AER): food aid $\Rightarrow$ conflict using
        interaction IV
  \item Acemoglu \& Restrepo (2020, JPE): robots $\Rightarrow$ employment
        loss using Bartik IV
\end{itemize}
\end{frame}
